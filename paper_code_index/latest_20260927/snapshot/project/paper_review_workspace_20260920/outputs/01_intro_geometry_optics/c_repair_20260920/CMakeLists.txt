cmake_minimum_required(VERSION 3.12)
project(laue511_repaired LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Geant4 11.4 REQUIRED)
include(${Geant4_USE_FILE})
add_executable(laue511 src/laue511.cc)
target_include_directories(laue511 PRIVATE src)
target_link_libraries(laue511 PRIVATE ${Geant4_LIBRARIES})
target_compile_options(laue511 PRIVATE -Wall -Wextra -Wpedantic)
add_executable(kernel_check src/kernel_check.cc)
target_include_directories(kernel_check PRIVATE src)
target_link_libraries(kernel_check PRIVATE ${Geant4_LIBRARIES})
